64static void run_rows(struct ge *ge, const struct msl_timing_chart *rows)
65{
66 const struct msl_timing_chart *chart;
67 uint32_t i = 0;
68
69 do {
70 const char *clock_name = ge_clock_name(ge->current_clock);
71 chart = &rows[i++];
72
73 if (chart->clock != ge->current_clock)
74 continue;
75
76 ge_print_registers_verbose(ge);
77
78 if (chart->additional) {
79 if (!chart->additional(ge)) {
80 ge_log(LOG_CONDS, " time %-4s - additional false\n", clock_name);
81 continue;
82 }
83 ge_log(LOG_CONDS, " time %-4s - additional true\n", clock_name);
84 }
85
86 if (chart->condition) {
87 if (!chart->condition(ge)) {
88 ge_log(LOG_CONDS, " time %-4s - condition false\n", clock_name);
89 continue;
90 }
91 ge_log(LOG_CONDS, " time %-4s - condition true\n", clock_name);
92 }
93
94
95 ge_log(LOG_CMDS, " %s\n", msl_comment_for_command(chart->command));
96 chart->command(ge);
97 } while (chart->clock < END_OF_STATUS);
98}

100void msl_run_state(struct ge* ge, const struct msl_timing_state *state)
101{
102 const struct msl_timing_chart *variant_rows;
103
104 /* Resolve the decode first so the "chart <name>" trace line is emitted
105 * before the rows it explains, but run the common rows first: they are
106 * the state's own, and the sheets print them ahead of the family rows
107 * they share a clock with. */
108 variant_rows = select_variant_chart(ge, state);
109
110 if (state->chart) {
111 if (state->variants && state->chart_ref && ge->current_clock == TO00)
112 ge_log(LOG_CONDS, " chart common (%s)\n", state->chart_ref);
113 run_rows(ge, state->chart);
114 }
115
116 if (variant_rows)
117 run_rows(ge, variant_rows);
118}
