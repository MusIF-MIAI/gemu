14void ge_init(struct ge *ge)
15{
16 memset(ge, 0, sizeof(*ge));
17 /* Strapped as the machine at Electric Dreams, per the physical card
18 * identification of 2026-07-21: 0618034Z reads PONT2N on the board, and
19 * 0618035V is electrically a PONT2N as well (different part code, same
20 * strap). E03 = PONT2N is the UCE 468 row of TAB.1 (2 usec, MAX), so
21 * the assumed model survives the re-identification; F03 = PONT2N flips
22 * TAB.2 to "interruption enabled on connector 3" (it was modelled as
23 * connector 4). E04/F04 are believed empty, pending a physical check.
24 *
25 * (2026-07-21: the F03 card was found and restuffed; all four option
26 * sockets hold the same 4-bridge card type, annotated PONT2N.
27 * 2026-07-22: the cards' own catalog drawing was found -- cp10F dwg
28 * 015 433 91 -- and settles the type: PONT 2N is a FOUR-bridge recipe
29 * (holes 3/20/21/25) shorting pins {1,2,4,7} to 17, PONT 2P likewise
30 * four bridges differing in one (pin 3 for pin 4). The in-machine
31 * cards match the 2N recipe, which retires the {1,3,4}-union reading;
32 * the pure-N strapping below stands. Final electrical seal: pin 1<->4
33 * beeps (2N) and pin 1<->3 does not. docs/hardware-options.md.)
34 *
35 * Card 05 follows the same both-are-PONT2N identification, which lands
36 * OFF the ch.001 table: the five printed rows stop at {N,P} = 32K and
37 * never define {N,N}. Under the card-pin mechanism (PONT2N shorts pins
38 * {1,4}, PONT2P shorts {1,3} -- signals.h) the N+N levels come out
39 * (VAMA2,VEMB6,VAMC2) = (1,0,0): no printed row, and the capacity is
40 * reported as off-table rather than guessed. What the memory bound
41 * logic does with (1,0,0) is an open question for the ch.077 consumers
42 * or for a memtest on the machine. See docs/hardware-options.md. */
43 ge->options.E03 = PONT_2N; /* UCE 468: 2 usec, MAX */
44 ge->options.F03 = PONT_2N; /* interruption enabled on connector 3 */
45 ge->options.F04 = PONT_NONE; /* empty = the interrupts-enabled variant */
46 ge->options.E04 = PONT_NONE; /* loading enabled on connectors 2 and 3 */
47 ge->options.E05 = PONT_2N; /* both 05 cards PONT2N: off-table combo */
48 ge->options.F05 = PONT_2N; /* -> (VAMA2,VEMB6,VAMC2) = (1,0,0) */
49
50 ge->ALTO = 1; /* stopped until CLEAR + START */
51 ge->powered = 1;
52 ge->register_selector = RS_NORM;
53
54 ge->ST3.name = "ST3";
55 ge->ST4.name = "ST4";
56
57 ge->channel2.name = "CAN2";
58}

60/*
61 * Resolve every strapped option and report it once, at CLEAR.
62 *
63 * The FUL / FEL / VAM / INES signals are not free-standing settings: each is
64 * a level produced by a jumper card in a backplane connector, so the only
65 * thing that is really configured is which cards are fitted where. This maps
66 * that to the signals the logic actually reads, which is what you want when
67 * comparing gemu against a physical machine -- the levels here should match
68 * what a meter reads on the corresponding connector pins.
69 *
70 * cp06 ch.001 (memory capacity) and ch.002 (version, loading, interruptions);
71 * see docs/hardware-options.md.
72 */
73static const char *pont_name(enum ge_pont p)
74{
75 switch (p) {
76 case PONT_2N: return "PONT2N";
77 case PONT_2P: return "PONT2P";
78 default: return "(empty)";
79 }
80}

173/* Seed the eight change / segment-base registers to their identity defaults
174 * N<<12: change register N is the 16-bit big-endian word at mem[240+2N], and
175 * an instruction address with modifier N (address bits 12-14) resolves to
176 * displacement + base[N]. With these defaults a bare 12-bit displacement
177 * carrying modifier N addresses segment N (0x1000*N ..), so a program's paged
178 * addresses (e.g. JU 0x172a) resolve to their full load addresses; programs
179 * may reload a base via LR/LA for paged access.
180 *
181 * Called by ge_clear (reset) and re-applied after a direct binary image load,
182 * because a contiguous image spanning the 0x00F0-0x00FF window would otherwise
183 * overwrite the bases (with its own bytes, or zeros in reconstructed gaps). */
184void ge_seed_segment_bases(struct ge *ge)
185{
186 for (int n = 0; n < 8; n++) {
187 uint16_t v = (uint16_t)(n << 12);
188 ge->mem[240 + 2 * n] = (uint8_t)(v >> 8);
189 ge->mem[240 + 2 * n + 1] = (uint8_t)(v & 0xff);
190 ge->cr_cache[n] = v; /* seed the addressing cache to match */
191 }
192}

207/* odd-parity bit for a byte: 1 if the byte has an even number of set bits
208 * (so data+parity is odd). Mirrors odd_parity() in pulse.c. */
209static inline uint8_t ge_odd_parity(uint8_t data)
210{
211 return __builtin_parity(data) ? 0 : 1;
212}
213
214/* Store a byte to memory the way every real write does: data + generated odd
215 * parity + mark written. The microcoded path (pulse.c on_TO65) does this for
216 * itself (and honours the INCE check-bit forcing); this helper is for the
217 * hybrid ALU/SS execution helpers (alu_*.c) and the change-register store,
218 * which write ge->mem[] directly and would otherwise leave stale parity and
219 * trip a false MEM CHECK on read-back. */
220void ge_mem_store8(struct ge *ge, uint16_t addr, uint8_t val)
221{
222 ge->mem[addr] = val;
223 ge->mem_parity[addr] = ge_odd_parity(val);
224 ge->mem_written[addr] = 1;
225}

227/* Load a flat image into memory at `origin` (the unified-format payload).
228 * Unlike ge_load_program this is origin-aware and not size-capped; it also
229 * primes the parity store and marks the cells written, so reads of the loaded
230 * code parity-check cleanly. Returns 0 on success, -1 if it would exceed the
231 * installed memory. */
232int ge_load_image(struct ge *ge, const uint8_t *image, size_t size,
233 uint16_t origin)
234{
235 uint32_t max = ge->mem_size ? ge->mem_size : MEM_SIZE;
236
237 if (image == NULL && size != 0)
238 return -1;
239 if ((uint32_t)origin + (uint32_t)size > max)
240 return -1;
241
242 for (size_t i = 0; i < size; i++) {
243 uint16_t a = (uint16_t)(origin + i);
244 ge->mem[a] = image[i];
245 ge->mem_parity[a] = ge_odd_parity(image[i]);
246 ge->mem_written[a] = 1;
247 }
248 return 0;
249}

251/* Enter execution at `entry` without the peripheral LOAD bootstrap: seed the
252 * program counter and drop the sequencer straight into the alpha (fetch) phase.
253 * Use after ge_clear + ge_load_image for the direct binary-load path; the
254 * --deck card-reader path uses the natural state 00 -> 80 -> alpha bootstrap
255 * instead (which leaves entry at 0). */
256void ge_enter(struct ge *ge, uint16_t entry)
257{
258 ge->rPO = entry;

394int ge_run_pulse(struct ge *ge)
395{
396 int r;
397 const struct msl_timing_state *state;
398
399 if (ge_clock_is_first(ge)) {
400 r = ge_peri_on_clock(ge);
401 if (r != 0)
402 return r;
403
404 /* poll the connectors and try to set up the cpu state.
405 * should this be here? */
406 connectors_first_clock(ge);
407 }
408
409 /* Execute common pulse machine logic */
410 pulse(ge);
411
412 /* Execute peripherals pulse callbacks */
413 r = ge_peri_on_pulses(ge);
414 if (r != 0)
415 return r;
416
417 /* Execute the commands from the timing charts */
418 state = msl_get_state(ge->rSA);
419
420 /* The state to execute gets loaded in SA at TO10 */
421 if (ge->current_clock == TO10)
422 ge_print_well_known_states(ge->rSA);
423
424 if (!state) {
425 ge_log(LOG_ERR, "no timing charts found for state %02X\n", ge->rSA);
426 return 1;
427 }
428
429 /* Latch the instruction-start PC for the disassembly display. In the alpha
430 * fetch (e2/e3) PO addresses the opcode and is NOT advanced within the state
431 * (operand fetch / PO recomputation happens in the later e0/e4/e6 states), so
432 * this is the address of the instruction now executing. It stays put while
433 * operands are read and PO is recomputed (e.g. across a jump), so a UI
434 * highlight tracking it does not drift onto operand (DB) bytes or the next
435 * line — it only moves when the next instruction is actually fetched. */
436 if (ge->rSA == 0xe2 || ge->rSA == 0xe3)
437 ge->instr_pc = ge->rPO;
438
439 msl_run_state(ge, state);
440
441 if (ge_clock_is_last(ge)) {
442 fsn_last_clock(ge);
443 ge_print_registers_nonverbose(ge);
444 }
445
446 ge_clock_increment(ge);
447 return 0;
448}

475void fsn_last_clock(struct ge *ge)
476{
477 /* At the end of a CPU cycle the future-status network is stored in SO
478 * (cpu fo. 127), advancing the program sequencer one state.
479 *
480 * In maintenance forcing (rotary off NORM) the program sequencer is frozen:
481 * the manual (CPU[4] §4 "Maintenance Panel", dwg 30004122 fo. 35-37) says a
482 * forcing cycle writes the *register under exam* (displayed through BO), it
483 * does not step the program. So a forcing cycle must NOT advance SO — that
484 * is what lets the operator key an instruction across phases (force SO=E2,
485 * step to E0, force FO, step to the 0x64 beta, force L1, step to execute).
486 * The one exception is rotary position 13 (RS_SO), which forces SO/SI
487 * itself — that is how the operator sets the sequencer state.
488 *
489 * RICI ("disable next status") suppresses the advance in normal operation,
490 * letting a status be re-executed. */
491 uint8_t sel_norm = ge->register_selector == RS_NORM;
492 uint8_t sel_so = ge->register_selector == RS_SO;
493 uint8_t advance_so = sel_norm ? !ge->console_switches.RICI : sel_so;
494 if (ge->RIA0 && advance_so) {
495 ge_log(LOG_FUTURE, "last clock cpu, %02x in SO\n", ge->future_state);
496 ge->rSO = ge->future_state;
497 } else {
498 ge_log(LOG_FUTURE, "last clock cpu, SO held at %02x (RIA0 %d advance %d)\n",
499 ge->rSO, ge->RIA0, advance_so);
500 }
501
502 /* after the end of a cpu work cycle, (ALTO / ALS71=1) is set if
503 * the PAPA switch is inserted, or if the rotary switch is neither
504 * in the normal position, nor in position 8 for recording in
505 * memory ALSOA=0) (cpu fo. 98)
506 */
507 uint8_t is_papa = ge->console_switches.PAPA;
508 uint8_t is_norm = ge->register_selector == RS_NORM;
509 uint8_t is_scr = ge->register_selector == RS_V1_SCR;
510
511 /* Step-by-step (PAPA, the ASIN request) can be inhibited by the program:
512 * INS sets ADIR=1, ENS / CLEAR clear it (CPU[4] §3.3). The maintenance STOC
513 * switch overrides the inhibit. This is the HW gate ALTO <- ... ASIN(ATOC +
514 * !ADIR), with ATOC = STOC (msl-states.c fo., cpu fo. 98). */
515 uint8_t step_enabled = ge->console_switches.STOC || !ge->ADIR;
516 uint8_t papa_stop = is_papa && step_enabled;
517 ge_log(LOG_FUTURE, " papa: %d, step_en: %d, norm: %d, scr: %d ==> %d\n",
518 is_papa, step_enabled, is_norm, is_scr,
519 ge->RIA0 && (papa_stop || !(is_norm || is_scr)));
520
521 if (ge->RIA0 && (papa_stop || !(is_norm || is_scr)))
522 ge->ALTO = 1;
523
524 /* PATE stops the timing after every cycle of the delay line — a finer step
525 * than PAPA and, unlike PAPA, it is not gated by the CPU/channel cycle
526 * (RIA0/RIA2), so it does interfere with peripheral transfers. One START
527 * then runs exactly one delay-line cycle. (CPU[4] §4, fo.35) */
528 if (ge->console_switches.PATE)
529 ge->ALTO = 1;
530
531 /* after the execution of a channel 2 cycle, load the first
532 * 4 bits of the future status network in SI. (cpu fo. 127) */
533 if (ge->RIA2) {
534 ge_log(LOG_FUTURE, "last clock ch2, %02x in SI\n", ge->future_state);
535 ge->rSI = ge->future_state;
536 }
