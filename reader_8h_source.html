10/*
11 * The integrated card reader on connector 2 / channel 1, modelled signal-by-
12 * signal against the COCA connector (see docs/signals.md). The data path
13 * (lu08/fini/data) is the working base; the remaining 1-bit fields are the
14 * other COCA pins, each a discrete modelled wire (reader->CPU status and
15 * CPU->reader command/mode). They are added inert and wired up phase by phase.
16 */
17struct ge_integrated_reader {
18 /* --- data path (base) --- */
19 uint8_t lu08:1; /* LU08N : character-ready strobe */
20 uint8_t fini:1; /* FININ : end-of-read, raised with the last char */
21 uint8_t data; /* LU00N-LU07N : the transcoded character byte */
22
23 /* --- reader -> CPU status lines --- */
24 uint8_t fiden:1; /* FIDEN : end-of-sequence (record fully read) */
25 uint8_t lupor:1; /* LUPOR : reader free / ready (= signal LUPO1) */
26 uint8_t luren:1; /* LUREN : error (transcoder / jam) */
27 uint8_t lusen:1; /* LUSEN : out-of-service */
28 uint8_t lenon:1; /* LENON : "not operable" condition (LENOB/LENO) */
29 uint8_t bi20:1; /* BI20 : binary-read 2nd-nibble aux clock */
30 uint8_t pom01:1; /* POM01 : binary-mode (by-pass) indicator */
31 uint8_t picon:1; /* PICON : first-column check */
32
33 /* --- CPU -> reader command / mode lines (latched on this side) --- */
34 uint8_t tu00:1; /* TU00N : read-strobe clock for RE data */
35 uint8_t tu03:1; /* TU03N : card-feed / advance clock */
36 uint8_t rifan:1; /* RIFAN : card reject / eject */
37 uint8_t regen:1; /* REGEN : general clear / reset */
38 uint8_t sesen:1; /* SESEN : put-in-manual */
39 uint8_t cocon:1; /* COCON : mode-select clock (latches the decode) */
40 uint8_t mode_n001:1;/* N001 : normal-mode decode */
41 uint8_t mode_n002:1;/* N002 */
42 uint8_t mode_debi:1;/* DEBI : binary-read (by-pass) decode */
43 uint8_t mode_mi01:1;/* MI01 : mixed-mode decode */
44 uint8_t mode_mi02:1;/* MI02 */
45
46 /* Read mode the CPU has selected, latched by COCON. While active_valid==0
47 * (the CPU has not driven a mode) the cardreader peripheral falls back to
48 * its harness-registered transcode mode; once the CPU issues a COCON latch
49 * active_mode takes over. */
50 enum transcode_mode active_mode;
51 uint8_t active_valid:1;
52};
