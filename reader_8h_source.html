10/*
11 * The integrated card reader on connector 2 / channel 1, modelled signal-by-
12 * signal against the COCA connector (see docs/signals.md). The data path
13 * (lu08/fini/data) is the working base; the remaining 1-bit fields are the
14 * other COCA pins, each a discrete modelled wire (reader->CPU status and
15 * CPU->reader command/mode). They are added inert and wired up phase by phase.
16 */
17struct ge_integrated_reader {
18 /* --- data path (base) --- */
19 uint8_t lu08:1; /* LU08N : character-ready strobe */
20 uint8_t fini:1; /* FININ : end-of-read, raised with the last char */
21 uint8_t data; /* LU00N-LU07N : the transcoded character byte */
22
23 /* --- reader -> CPU status lines --- */
24 uint8_t fiden:1; /* FIDEN : end-of-sequence (deck exhausted), L1·03 */
25 uint8_t lupor:1; /* LUPOR : reader free / ready, L1·07 (= LUPO1) */
26 uint8_t luren:1; /* LUREN : error (transcoder / jam), L1·09 */
27 uint8_t lusen:1; /* LUSEN : out-of-service, L1·04 (strap) */
28 uint8_t lenon:1; /* LENON : "not operable", L1·13 (strap) */
29 uint8_t lesab:1; /* LESAB : reader present / selected, L1·15 (strap)*/
30 uint8_t pom01:1; /* POM01 : binary-mode (by-pass) indicator, M1·12 */
31
32 /* These two are real reader wires but live on the harness sub-connectors
33 * (BI20 on 1253-2, PICON on 1251·24), NOT on the COCA slots I1/L1/M1 that
34 * carry everything above. A board attaching at the COCA slots -- like the
35 * Pico reader -- never sees them, which is why the firmware has no
36 * equivalent. Modelled here for observability. */
37 uint8_t bi20:1; /* BI20 : binary-read 2nd-nibble aux clock */
38 uint8_t picon:1; /* PICON : first-column check */
39
40 /* --- CPU -> reader command / mode lines (latched on this side) --- */
41 uint8_t tu00:1; /* TU00N : read-strobe clock for RE data */
42 uint8_t tu03:1; /* TU03N : card-feed / advance clock */
43 uint8_t rifan:1; /* RIFAN : card reject / eject */
44 uint8_t regen:1; /* REGEN : general clear / reset */
45 uint8_t sesen:1; /* SESEN : put-in-manual */
46 uint8_t cocon:1; /* COCON : mode-select clock (latches the decode) */
47 uint8_t mode_n001:1;/* N001 : normal-mode decode */
48 uint8_t mode_n002:1;/* N002 */
49 uint8_t mode_debi:1;/* DEBI : binary-read (by-pass) decode */
50 uint8_t mode_mi01:1;/* MI01 : mixed-mode decode */
51 uint8_t mode_mi02:1;/* MI02 */
52
53 /* Read mode the CPU has selected, latched by COCON. While active_valid==0
54 * (the CPU has not driven a mode) the cardreader peripheral falls back to
55 * its harness-registered transcode mode; once the CPU issues a COCON latch
56 * active_mode takes over. */
57 enum transcode_mode active_mode;
58 uint8_t active_valid:1;
59
60 /* --- command latch (rpi-pico-card-reader src/feeder.c) ---
61 *
62 * A read command does NOT strobe anything. It is latched here; the card is
63 * presented when the CPU's TU03N feed arrives, which on the bench is the
64 * order the wire actually shows (command, then feed, then strobes). Feeding
65 * on the command instead loses the card to a channel that is not armed yet.
66 */
67 uint8_t cmd_pending:1; /* a read command is latched, awaiting the feed */
68 uint8_t cmd_reject:1; /* 0x48 card reject: drop this card, no transfer */
69
70 /* Commands the reader does not implement. Counted rather than acted on, so
71 * an unexpected order shows up in a trace instead of disturbing the feed. */
72 uint16_t anomalies;
73};
