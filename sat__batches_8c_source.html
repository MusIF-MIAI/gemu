47/* Printer Mechanic Test startup: the deck issues PER 0x00,0x0126, whose order
48 * block is {00,40,00,4F,06,70}. LL is a length-1 field, so 0x4F reads 80 bytes —
49 * exactly one 80-column card — "unchanged" into 0x0670. In the
50 * real SAT stack that record comes from the required center card, which the
51 * operator places behind the program deck. So punch one: a row-0 hole is 0x01
52 * under the raw ("read unchanged") reader, a blank column is 0x00.
53 *
54 * column 1: integrated subsystem -> 0x01
55 * column 2: 2nd transport absent -> 0x01
56 * column 3: (not evidenced) -> blank
57 * column 4: normal drum -> 0x01
58 * column 5: normal ribbon -> 0x01
59 * column 6: with END OF TEST HLT -> blank (a 9-punch collapses to 0x00
60 * under the raw low-byte model)
61 *
62 * Confidence: 0x0673 / 0x0674 (columns 4 and 5) are directly tested by the deck
63 * as 0x01. Column 6 follows the manual. Columns with no evidence stay blank. */
64static const uint16_t printer_mech_center_card[80] = {
65 0x0001, 0x0001, 0x0000, 0x0001, 0x0001, 0x0000,
66 /* columns 7..80 blank */
67};

138/* The shared loader-card rule (rpi-pico-card-reader src/deck.c
139 * deck_find_loader_card): marker = a row-8 punch in column 3, searched over
140 * card indices 0..4 inclusive, first match wins; with no marker fall back to
141 * card 1 on a multi-card deck and card 0 otherwise. Card 0 must be in the
142 * window: captured box decks lead with a title card, but a synthesized deck
143 * (gasm --boot/--bootge) leads with the loader itself. */
144static int row8_loader_card(const struct cap_deck *deck)
145{
146 int ncards = cap_num_cards(deck);
147 int limit = ncards < 5 ? ncards : 5;
148
149 for (int i = 0; i < limit; i++) {
150 const uint16_t *cols = cap_card_columns(deck, i);
151 if (cap_card_ncols(deck, i) >= 3 && cols && cols[2] == 0x0100)
152 return i;
153 }
154 return (ncards > 1) ? 1 : 0;
155}

157static int append_card_range(struct cap_deck *out, const struct cap_deck *src,
158 int first, int last_inclusive)
159{
160 for (int i = first; i <= last_inclusive; i++) {
161 int ncols = cap_card_ncols(src, i);
162 const uint16_t *cols = cap_card_columns(src, i);
163 if (ncols <= 0 || !cols)
164 continue;
165 if (cap_append_card(out, cols, ncols) != 0)
166 return -1;
167 }
168 return 0;
169}

171static int append_source(struct cap_deck *out, const char *path, enum sat_source_op op)
172{
173 struct cap_deck *src = cap_load(path);
174 int ncards;
175 int rc = -1;
176
177 if (!src)
178 return -1;
179
180 ncards = cap_num_cards(src);
181 switch (op) {
182 case SAT_SRC_AS_IS:
183 rc = append_card_range(out, src, 0, ncards - 1);
184 break;
185 case SAT_SRC_TRIM_TITLE_SUMMARY:
186 rc = (ncards >= 3) ? append_card_range(out, src, 1, ncards - 2) : -1;
187 break;
188 case SAT_SRC_SERIAL_LOADER_PLUS_BODY: {
189 int loader = row8_loader_card(src);
190 if (ncards < 7)
191 rc = -1;
192 else if (cap_append_card(out, cap_card_columns(src, loader),
193 cap_card_ncols(src, loader)) != 0)
194 rc = -1;
195 else
196 rc = append_card_range(out, src, 5, ncards - 2);
197 break;
198 }
199 default:
200 rc = -1;
201 break;
202 }
203
204 cap_free(src);
205 return rc;
206}

244int sat_batch_prepare_deck(const char *root, const char *id,
245 const char *out_path,
246 char *note, size_t note_sz)
247{
248 char path[1024];
249 struct cap_deck *out;
250 const struct sat_batch_def *def = sat_batch_def_find(id);
251
252 if (!def || !root || !out_path)
253 return -1;
254
255 out = cap_create();
256 if (!out)
257 return -1;
258
259 for (int i = 0; i < def->nsources; i++) {
260 snprintf(path, sizeof(path), "%s/%s", root, def->sources[i].file);
261 if (append_source(out, path, def->sources[i].op) != 0) {
262 cap_free(out);
263 return -1;
264 }
265 }
266
267 if (def->tail && cap_append_card(out, def->tail->cols, def->tail->ncols) != 0) {
268 cap_free(out);
269 return -1;
270 }
271
272 if (cap_save(out, out_path) != 0) {
273 cap_free(out);
274 return -1;
275 }
276
277 cap_free(out);
278 sat_note(note, note_sz, def);
279 return 0;
280}
